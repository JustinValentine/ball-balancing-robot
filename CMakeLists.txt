cmake_minimum_required(VERSION 3.16)
project(ball_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DC1394 REQUIRED IMPORTED_TARGET libdc1394-2)
add_executable(ball-vision src/main.cpp src/firewire.cpp src/tracker.cpp)
target_link_libraries(ball-vision PRIVATE ${OpenCV_LIBS} PkgConfig::DC1394)
target_compile_options(ball-vision PRIVATE -Wall -Wextra -Wpedantic)
